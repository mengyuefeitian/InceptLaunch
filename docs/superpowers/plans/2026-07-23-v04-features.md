# InceptLaunch v0.4 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Fix the single-tap-dismiss bug (4th report) and ship four features: grid tidy, folder enlarge/shrink, Liquid Glass materials, and runtime icon switching.

**Architecture:** The bug fix is a one-line gesture change in ContentView. Grid tidy exposes the existing `compactPages()` via a new ViewModel method triggered by a context menu. Folder enlarge adds `enlargedFolderIDs` to the layout model and a custom 2×2 tile view with 3×3 carousel. Liquid Glass applies macOS 26 `glassEffect` with material fallback. Icon switching stores a preference and sets `NSApp.applicationIconImage` at runtime.

**Tech Stack:** Swift 6.3, SwiftUI + AppKit, Swift Testing, macOS 26 SDK (glass API with availability guard)

```swift
import SwiftUI

/// A 2×2 folder tile showing members in a 3×3 grid with carousel paging
/// when there are more than 9 members.
struct EnlargedFolderTileView: View {
    let item: LaunchpadDisplayItem
    let iconSize: CGFloat
    let tileWidth: CGFloat
    let tileHeight: CGFloat
    let onLaunch: (AppRecord) -> Void

    @State private var currentPage = 0

    private var pages: [[AppRecord]] {
        let members = item.members
        guard members.count > 9 else { return [members] }
        return stride(from: 0, to: members.count, by: 9).map {
            Array(members[$0..<min($0 + 9, members.count)])
        }
    }

    var body: some View {
        VStack(spacing: 6) {
            TabView(selection: $currentPage) {
                ForEach(pages.indices, id: \.self) { pageIndex in
                    memberGrid(pages[pageIndex])
                        .tag(pageIndex)
                }
            }
            .tabViewStyle(.page(indexDisplayMode: pages.count > 1 ? .automatic : .never))

            Text(item.title)
                .font(.callout)
                .lineLimit(1)
                .foregroundStyle(.white)
        }
        .frame(width: tileWidth * 2 + GridMetrics.columnSpacing,
               height: tileHeight * 2 + GridMetrics.rowSpacing)
        .background(
            RoundedRectangle(cornerRadius: 28, style: .continuous)
                .fill(.white.opacity(0.12))
        )
        .contentShape(Rectangle())
    }

    private func memberGrid(_ members: [AppRecord]) -> some View {
        let cols = Array(repeating: GridItem(.fixed(iconSize * 0.7), spacing: 8), count: 3)
        return LazyVGrid(columns: cols, spacing: 8) {
            ForEach(members) { member in
                RealAppIcon(record: member)
                    .frame(width: iconSize * 0.7, height: iconSize * 0.7)
                    .onTapGesture { onLaunch(member) }
            }
        }
        .padding(12)
    }
}
```

```swift
import SwiftUI

/// Applies macOS 26 Liquid Glass when available, falls back to material.
struct GlassBackground: ViewModifier {
    var cornerRadius: CGFloat = 24
    var style: GlassStyle = .regular

    enum GlassStyle { case regular, thick }

    func body(content: Content) -> some View {
        if #available(macOS 26.0, *) {
            content
                .glassEffect(.regular.tint(.white.opacity(0.1)), in: .rect(cornerRadius: cornerRadius))
        } else {
            content
                .background(
                    RoundedRectangle(cornerRadius: cornerRadius, style: .continuous)
                        .fill(style == .thick ? AnyShapeStyle(.ultraThickMaterial) : AnyShapeStyle(.ultraThinMaterial))
                )
        }
    }
}

extension View {
    func glassBackground(cornerRadius: CGFloat = 24, style: GlassBackground.GlassStyle = .regular) -> some View {
        modifier(GlassBackground(cornerRadius: cornerRadius, style: style))
    }
}
```

```swift
import AppKit

/// Switches the app's Dock/menu-bar icon at runtime.
/// Note: only affects the running process; Finder still shows the bundle icon.
@MainActor
enum IconSwitcher {
    static func apply(style: String) {
        let iconName: String
        switch style {
        case "A": iconName = "IconA"
        case "B": iconName = "IconB"
        default:  iconName = "InceptLaunch"  // D = default bundle icon
        }

        if style == "D" {
            // Reset to bundle icon
            NSApp.applicationIconImage = NSImage(named: NSImage.applicationIconName)
        } else {
            let bundlePath = Bundle.main.bundlePath
            let icnsPath = "\(bundlePath)/Contents/Resources/Icons/\(iconName).icns"
            if let image = NSImage(contentsOfFile: icnsPath) {
                NSApp.applicationIconImage = image
            }
        }
    }
}
```
